Day 03 - Ansible, Terraform & Kubernetes Basics
------------------------------------------------

Ansible - Install Apache and start Service
------------------------------------------

name: Install & Start Apache
hosts: all
become: yes
tasks:
    name: Install Apache
    yum:
        name: httpd
        state: present
    
    name: Start Service
    services:
        name: httpd
        state: started
        enable: yes

Kubernetes - create a ReplicaSet with 3 Replicas
--------------------------------------------------

apiversion: api/v1
kind: ReplicaSet
metadata: 
    name: userName-ReplicaSet
Spec: 
    replicas: 3
    selector: 
        matchLabels:
        app: own-app
    template:
        metadata:
            labels:
                app: own-app
        spec:
            containers:
                name: app
                image: nginx
                port:
                    containerPort:80
