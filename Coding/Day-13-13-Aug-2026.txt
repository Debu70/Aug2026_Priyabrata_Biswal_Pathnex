Ansible : Setup Docker container for Nginx
-------------------------------------------

- name: setup Docker container for nginx
  hosts: all
  become: yes
  tasks: 
    - name: pull nginx Image
     docker_image:
        name: nginx
        source: pull
    - name: Run nginx container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published:_ports:
            - "8080:80"

Kubernetes - Horaizontal Pod AutoScalar(HPA)
--------------------------------------------

apiVersion : apps/v1.0
kind: deployment
metadata:
    name: own-deployment
spec:
    replicas: 1
    selector: 
        matchLabels:
        app: own-app
    template:
        metadata:
            lables:
                app: own-app
        spec:
            containers:
                - name: nginx
                  image: nginx
                  ports:
                    - containerPort: 80
---

apiVersion: autoscalling/v2.0
kind: HoraizontalAutoScaler
metadata:
    name: own-app
spec: 
    scaleTargetRef:
        apiVersion : app/v1.0
        kind: deployment
        name: own-deployment
    minReplicas: 1
    maxReplicas: 10
    targetPutilizationPercentage: 50

Jenkinfiles: Deploy to Kubernetes
-----------------------------------

pipeline{
    agent any
    stages{
        stage('Deploy to Kubernetes')
            step{
                script{
                    sh 'kubctl apply -f deployment.yaml'
            }
        }
    }
}
