Ansible - Deploy AWS Lambda Funtion Using AWS CLI
-------------------------------------------------
- name: Deploy AWS Lamda Funtion
  hosts: localhost
  tasks:
    - name: Create Lambda Funtion function
      command: aws lambda create-funtion --funtion-name own-lambda \
               --runtime python3.0 \
               --role arn:aws:iam::123456789:role/lambda-execution-role \
               --handler lambda_funtion.lambda_handler \
               --zip-file fileb://lambda.zip

Terraform - ServerLess API Gateway with Lambda Integration
----------------------------------------------------------
resource "aws_api_gateway_rest_api" "own-api"{
    name = "own-api"
    description = "own api" 
}

resource "aws_lambda_funtion" "own_lambda"{
    funtion_name = "own-lambda"
    runtime = "python3.0"
    role = aws_iam_role.lambda_exec_role.arn
    handler = "lambda_funtion.lambda_handler"
    fileName = "lambda.zip"
}

resource "aws_api_gateway_integration" "lambda_intigration"{
    rest_api_id = aws_api_gateway_rest_api.own_api.id
    resource_id = aws_api_gateway_resource.own_resource.id
    http_method = aws_api_gateway_method.own_method.http_method
    Integration_http_method = "POST"
    type = "AWS_PROXY"
    uri = aws_lambda_function.own_lambda.invoke_arn
}

Jenkins - Deploy ServerLess Application with lambda
---------------------------------------------------
pipeline{
    agent any
    stages{
        stage('Build'){
            steps{
                echo 'Building Lambda funtion....'
                sh 'zip -r lambda.zip lambda/'
            }
        }
        stage('Deploy'){
            steps{
                script{
                    sh 'aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip'
                }
            }
        }
    }
}

Multi Stage Docker Build-
------------------------

-> Stage - 1 Build Stage
-------------------------

From maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/own/java-build
COPY . /opt/own/java-build/
RUN mvn clean package

-> Stage - 2 Production Stage
-------------------------

FROM tomcat:9
WORKDIR /usr/local/tomcat/webapps
COPY --from-builder opt/own/java-build/target/*.war /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh", "run"]
